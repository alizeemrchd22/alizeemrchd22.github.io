Part 1 — Trending data (10 CSV files)

Part1_trending_csv.sql

-- ===== TRENDING DATA (the 10 csv files) =====
 
-- how to read the csv: comma separated, skip the header row
CREATE OR REPLACE FILE FORMAT file_format_csv
TYPE = 'CSV'
FIELD_DELIMITER = ','
SKIP_HEADER = 1
NULL_IF = ('\\N', 'NULL', 'NUL', '')
FIELD_OPTIONALLY_ENCLOSED_BY = '"'
;
 
-- external table = just reads straight from the stage, nothing stored yet
-- (checked the column order first by looking at the raw header row
-- before writing this, that's how i got c1..c11 right)
CREATE OR REPLACE EXTERNAL TABLE ex_table_youtube_trending
(
    video_id         varchar        as (value:c1::varchar),
    title            varchar        as (value:c2::varchar),
    publishedat      timestamp_ntz  as (value:c3::timestamp_ntz),
    channelid        varchar        as (value:c4::varchar),
    channeltitle     varchar        as (value:c5::varchar),
    categoryid       int            as (value:c6::int),
    trending_date    date           as (value:c7::date),
    view_count       int            as (value:c8::int),
    likes            int            as (value:c9::int),
    dislikes         int            as (value:c10::int),
    comment_count    int            as (value:c11::int)
)
WITH LOCATION = @stage_assignment
FILE_FORMAT = file_format_csv
PATTERN = '.*youtube_trending_data[.]csv';
 
-- now actually build the real table. also grabbing the country
-- from the filename (BR_youtube_trending_data.csv -> BR) since
-- all 10 countries get stacked into this one table
CREATE OR REPLACE TABLE table_youtube_trending AS
SELECT
    video_id,
    title,
    publishedat,
    channelid,
    channeltitle,
    categoryid,
    trending_date,
    view_count,
    likes,
    dislikes,
    comment_count,
    split_part(metadata$filename, '_', 1)::varchar AS country
FROM ex_table_youtube_trending;

Part 1 — Category data (10 JSON files)

part1_category_json.sql

-- ===== CATEGORY DATA (the 10 json files) =====
-- each file looks like { "items": [ {"id": "10", "snippet": {"title": "Music"}}, ... ] }
-- can't read this as columns like the csv, need to unfold the array first
 
CREATE OR REPLACE EXTERNAL TABLE ex_table_youtube_category
WITH LOCATION = @stage_assignment
FILE_FORMAT = (TYPE = JSON)
PATTERN = '.*category_id[.]json';
 
-- LATERAL FLATTEN unfolds the items array so each category becomes
-- its own row, same trick as before for getting country from filename
CREATE OR REPLACE TABLE table_youtube_category AS
SELECT
    split_part(metadata$filename, '_', 1)::varchar AS country,
    item.value:id::int                 AS categoryid,
    item.value:snippet:title::varchar  AS category_title
FROM ex_table_youtube_category,
LATERAL FLATTEN(input => value:items) item;
 
SELECT * FROM table_youtube_category LIMIT 10;
SELECT COUNT(*) FROM table_youtube_category;
SELECT DISTINCT country FROM table_youtube_category;

Part 1 — Final table

part1_table_youtube.sql

-- ===== FINAL TABLE =====
-- combining the two. using LEFT JOIN so i don't lose any trending
-- video even if a categoryid doesn't have a match (this actually
-- comes up as a question in part 2)
CREATE OR REPLACE TABLE table_youtube_final AS
SELECT
    UUID_STRING()      AS id,
    t.video_id,
    t.title,
    t.publishedat,
    t.channelid,
    t.channeltitle,
    t.categoryid,
    c.category_title,
    t.trending_date,
    t.view_count,
    t.likes,
    t.dislikes,
    t.comment_count,
    t.country
FROM table_youtube_trending t
LEFT JOIN table_youtube_category c
    ON t.country    = c.country
   AND t.categoryid = c.categoryid;

-- Create a new table called “table_youtube_duplicates”  containing only the “bad” duplicates by using the row_number() function.


-- these aren't exact duplicate rows, video_id + country + trending_date
-- are the same but view_count/likes/dislikes/comment_count are different.
-- the brief says to keep the one with the highest view_count so we keep
-- rn equal to 1 (the highest view_count) for the official table and rn
-- greater than 1 for the duplicates table.

SELECT *
FROM table_youtube_final
QUALIFY COUNT(*) OVER (PARTITION BY video_id, country, trending_date) > 1
ORDER BY video_id, country, trending_date, view_count DESC;

-- the result of this query show that we have 74932 duplicates which I thought was a lot. But after reading the brief we know that we have to achieve a final number of row of 2,597,494.
-- So i tryed some calculation to check and 74932 / 2 = 37,466, so because we know that our actual number of lines is around 2,634,960 (after deleting the #NAME? rows in the previous step), 
-- removing 37,466 bad duplicates gets us to exactly 2,597,494. Same number as 74932/2, so the math lines up before even running the delete.

SELECT *,
    ROW_NUMBER() OVER (
        PARTITION BY video_id, country, trending_date
        ORDER BY view_count DESC
    ) AS rn
FROM table_youtube_final
QUALIFY COUNT(*) OVER (PARTITION BY video_id, country, trending_date) > 1
ORDER BY video_id, country, trending_date, view_count DESC;

CREATE OR REPLACE TABLE table_youtube_duplicates AS
SELECT *
FROM table_youtube_final
QUALIFY ROW_NUMBER() OVER (
    PARTITION BY video_id, country, trending_date
    ORDER BY view_count DESC
) > 1;

SELECT COUNT(*) FROM table_youtube_duplicates;


-- Delete the duplicates in “table_youtube_final“ by using “table_youtube_duplicates”.

DELETE FROM table_youtube_final
WHERE id IN (SELECT id FROM table_youtube_duplicates);

-- Count the number of rows in “table_youtube_final“ and check that it is equal to 2,597,494 rows.

SELECT COUNT(*) FROM table_youtube_final;

-- PART 3 : Data ANALYSIS 
USE DATABASE assignment_1;

-- What are the 3 most viewed videos for each country in the Gaming category for the trending_date = '2024-04-01'. Order the result by country and the rank.
SELECT *
FROM table_youtube_final

  AND trending_date = '2024-04-01';

SELECT

    view_count,
    ROW_NUMBER() OVER (PARTITION BY country ORDER BY view_count DESC) AS rank

WHERE category_title = 'Gaming'
  AND trending_date = '2024-04-01';

-- NOw I only keep rank 1 to 3 per country
SELECT
    country,
    title,
    view_count,
    ROW_NUMBER() OVER (PARTITION BY country ORDER BY view_count DESC) AS rank
FROM table_youtube_final
WHERE category_title = 'Gaming'
  AND trending_date = '2024-04-01'
QUALIFY rank <= 3
ORDER BY country, rank;


-- For each country, count the number of distinct video with a title containing the word "BTS" (case insensitive) and order the result by count in a descending order

-- My reasonning: first the ilike '%bts%' serve to finds "bts" anywhere in the title, in a not case sensitive way. Then  we have to use count(distinct video_id) instead of count(*) because some videos can have many lines in a data set, so we want only the numb of videos and not lines. 
-- Then we will group then by county. 

SELECT *
FROM table_youtube_final
WHERE title ILIKE '%BTS%';

SELECT
    country,
    COUNT(DISTINCT video_id) AS nb_videos
FROM table_youtube_final
WHERE title ILIKE '%BTS%'
GROUP BY country
ORDER BY nb_videos DESC;


-- For each country, year and month (in a single column) and only for the year 2024, which video is the most viewed and what is its likes_ratio (defined as the percentage of likes against view_count) truncated to 2 decimals. Order the result by year_month and country.
-- My reasonnning: first I will filter from wide to specific, so I filter by the year_month column and select only year 2024
SELECT
    country,
    trending_date,
    DATE_TRUNC('month', trending_date) AS year_month,
    title,
    view_count,
    likes
FROM table_youtube_final
WHERE YEAR(trending_date) = 2024;

-- Then I apply the same column as in the brief exemple and make sure the ratio for the likes is based on the number of views

SELECT
    country,                                    -- COUNTRY
    DATE_TRUNC('month', trending_date) AS year_month,  -- YEAR_MONTH
    title,                                       -- TITLE
    channeltitle,                                -- CHANNELTITLE
    category_title,                              -- CATEGORY_TITLE
    view_count,                                  -- VIEW_COUNT
    TRUNCATE(likes / view_count * 100, 2) AS likes_ratio  -- LIKES_RATIO
FROM table_youtube_final
WHERE YEAR(trending_date) = 2024
QUALIFY ROW_NUMBER() OVER (
    PARTITION BY country, DATE_TRUNC('month', trending_date)
    ORDER BY view_count DESC
) = 1
ORDER BY year_month, country;


-- For each country, which category_title has the most distinct videos and what is its percentage (2 decimals) out of the total distinct number of videos of that country? Only look at the data from 2022. Order the result by category_title and country.
-- MY reasoning: this question can be solved in two step, first we are asked to find which category as the most videos ( in video ID not lines in the dataset) so I will use the COUNT(DISTINCT video_id). 

SELECT
    country,
    category_title,
    COUNT(DISTINCT video_id) AS nb_videos
FROM table_youtube_final
WHERE YEAR(trending_date) = 2022
GROUP BY country, category_title
ORDER BY country, nb_videos DESC;

-- Then I need to find the TOtal of videos all categories per country to get the denominator for the percentage 

SELECT
    country,
    COUNT(DISTINCT video_id) AS total_videos
FROM table_youtube_final
WHERE YEAR(trending_date) = 2022
GROUP BY country;

-- I can finally do a final block
WITH cat_counts AS (
    SELECT
        country,
        category_title,
        COUNT(DISTINCT video_id) AS nb_videos
    FROM table_youtube_final
    WHERE YEAR(trending_date) = 2022
    GROUP BY country, category_title
),
country_totals AS (
    SELECT
        country,
        COUNT(DISTINCT video_id) AS total_videos
    FROM table_youtube_final
    WHERE YEAR(trending_date) = 2022
    GROUP BY country
)
SELECT
    c.country,
    c.category_title,
    c.nb_videos AS total_category_video,
    t.total_videos AS total_country_video,
    TRUNCATE(c.nb_videos / t.total_videos * 100, 2) AS percentage
FROM cat_counts c
JOIN country_totals t ON c.country = t.country
QUALIFY ROW_NUMBER() OVER (PARTITION BY c.country ORDER BY c.nb_videos DESC) = 1
ORDER BY c.category_title, c.country;


-- Which channeltitle has produced the most distinct videos and what is this number*100? 

-- in a more simple way, the question is to find which  youtube channel has the most different like uniquevideos (using video id and not lines)  in the whole dataset (all countries, all dates combined), and how many videos that is.

SELECT

    COUNT(DISTINCT video_id) AS nb_videos

SELECT
    YEAR(trending_date) AS year,
    MIN(trending_date) AS first_date,
    MAX(trending_date) AS last_date,
    COUNT(DISTINCT trending_date) AS nb_days_covered
FROM table_youtube_final
GROUP BY YEAR(trending_date)
ORDER BY year;

-- now we need to study the trend prpgression only on full years
SELECT
    category_title,
    YEAR(trending_date) AS year,
    COUNT(DISTINCT video_id) AS nb_videos
FROM table_youtube_final
WHERE category_title IN ('Sports', 'People & Blogs', 'Gaming', 'Comedy')
  AND YEAR(trending_date) IN (2021, 2022, 2023)
GROUP BY category_title, YEAR(trending_date)
ORDER BY category_title, year;

WITH yearly AS (
    SELECT
        category_title,
        YEAR(trending_date) AS year,
        COUNT(DISTINCT video_id) AS nb_videos
    FROM table_youtube_final
    WHERE category_title IN ('Sports', 'People & Blogs', 'Gaming', 'Comedy')
      AND YEAR(trending_date) IN (2021, 2022, 2023)
    GROUP BY category_title, YEAR(trending_date)
)
SELECT
    category_title,
    year,
    nb_videos,
    LAG(nb_videos) OVER (PARTITION BY category_title ORDER BY year) AS prev_year,
    TRUNCATE(
        (nb_videos - LAG(nb_videos) OVER (PARTITION BY category_title ORDER BY year))
        / LAG(nb_videos) OVER (PARTITION BY category_title ORDER BY year) * 100,
        2
    ) AS pct_change
FROM yearly
ORDER BY category_title, year;

-- I decided to normalize by days covered, to compare 2023 vs the partial 2024 properly ( we can now confirm that gamin has the most decline and People & Blogs seem more stable )
WITH yearly AS (
    SELECT
        category_title,
        YEAR(trending_date) AS year,
        COUNT(DISTINCT video_id) AS nb_videos,
        COUNT(DISTINCT trending_date) AS nb_days,
        TRUNCATE(COUNT(DISTINCT video_id) / COUNT(DISTINCT trending_date), 2) AS videos_per_day
    FROM table_youtube_final
    WHERE category_title IN ('Sports', 'People & Blogs', 'Gaming', 'Comedy')
      AND YEAR(trending_date) IN (2023, 2024)
    GROUP BY category_title, YEAR(trending_date)
)
SELECT
    category_title,
    year,
    videos_per_day,
    LAG(videos_per_day) OVER (PARTITION BY category_title ORDER BY year) AS prev_rate,
    TRUNCATE(
        (videos_per_day - LAG(videos_per_day) OVER (PARTITION BY category_title ORDER BY year))
        / LAG(videos_per_day) OVER (PARTITION BY category_title ORDER BY year) * 100,
        2
    ) AS pct_change_rate
FROM yearly
ORDER BY category_title, year;

-- here i am getting the top 5 of the channels that had the most videos in trends per category 

SELECT
    category_title,
    channeltitle,
    COUNT(DISTINCT video_id) AS nb_videos
FROM table_youtube_final
WHERE category_title IN ('Sports', 'People & Blogs', 'Gaming', 'Comedy')
GROUP BY category_title, channeltitle
QUALIFY ROW_NUMBER() OVER (PARTITION BY category_title ORDER BY COUNT(DISTINCT video_id) DESC) <= 5
ORDER BY category_title, nb_videos DESC;

-- now I calculate what this portion represent with a % 

WITH cat_totals AS (
    SELECT
        category_title,
        COUNT(DISTINCT video_id) AS total_videos
    FROM table_youtube_final
    WHERE category_title IN ('Sports', 'People & Blogs', 'Gaming', 'Comedy')
    GROUP BY category_title
),
top5_channels AS (
    SELECT
        category_title,
        channeltitle,
        COUNT(DISTINCT video_id) AS nb_videos
    FROM table_youtube_final
    WHERE category_title IN ('Sports', 'People & Blogs', 'Gaming', 'Comedy')
    GROUP BY category_title, channeltitle
    QUALIFY ROW_NUMBER() OVER (PARTITION BY category_title ORDER BY COUNT(DISTINCT video_id) DESC) <= 5
),
top5_sum AS (
    SELECT
        category_title,
        SUM(nb_videos) AS top5_total
    FROM top5_channels
    GROUP BY category_title
)
SELECT
    t.category_title,
    s.top5_total,
    t.total_videos,
    TRUNCATE(s.top5_total / t.total_videos * 100, 2) AS concentration_pct
FROM top5_sum s
JOIN cat_totals t ON s.category_title = t.category_title
ORDER BY concentration_pct DESC;

-- now I take a closer look to People & Blogs to see it's global performance across different countries

SELECT

----- POPULARITY ANALYSIS ------

-- The goal of this section is to know the turnover of videos being top trends, so for each video, in each country, how many different days it stays trending
SELECT
    category_title,
    video_id,
    country,
    COUNT(DISTINCT trending_date) AS nb_days_trending
FROM table_youtube_final
WHERE category_title IN ('Sports', 'People & Blogs', 'Gaming', 'Comedy')
GROUP BY category_title, video_id, country;


-- I compute the average per category because for the lower = more turnover = more new opportunities cycling in
WITH video_country_days AS (
    SELECT
        category_title,
        video_id,
        country,
        COUNT(DISTINCT trending_date) AS nb_days_trending
    FROM table_youtube_final
    WHERE category_title IN ('Sports', 'People & Blogs', 'Gaming', 'Comedy')
    GROUP BY category_title, video_id, country
)
SELECT
    category_title,
    TRUNCATE(AVG(nb_days_trending), 2) AS avg_days_trending_per_video
FROM video_country_days
GROUP BY category_title
ORDER BY avg_days_trending_per_video ASC;

A category could be easy to break into but with pretty low views once you're there. This would add to the recommendation: not just "it's accessible" but "it's accessible AND gets a decent audience"

SELECT
